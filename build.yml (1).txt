name: Build APK

on:
  push:
    branches: [builds]

permissions:
  contents: write
  actions: read

jobs:
  build:
    runs-on: ubuntu-latest
    steps:

      - name: Préparation
        uses: actions/checkout@v4
        with:
          ref: builds

      - name: Extraction
        run: |
          set -e
          mkdir -p project
          if ! unzip -q builds/incoming.zip -d project; then
            echo "::error::L'archive builds/incoming.zip est invalide ou corrompue."
            exit 1
          fi

          # Si le zip contient un unique dossier racine, on s'aligne dessus
          entries=(project/*)
          if [ "${#entries[@]}" -eq 1 ] && [ -d "${entries[0]}" ]; then
            shopt -s dotglob
            mv "${entries[0]}"/* project/ 2>/dev/null || true
            shopt -u dotglob
          fi

          MANIFEST_OK=0
          if [ -f project/AndroidManifest.xml ] || [ -f project/app/src/main/AndroidManifest.xml ]; then
            MANIFEST_OK=1
          fi
          BUILD_GRADLE_OK=0
          if [ -f project/build.gradle ] || [ -f project/build.gradle.kts ]; then
            BUILD_GRADLE_OK=1
          fi
          SETTINGS_OK=0
          if [ -f project/settings.gradle ] || [ -f project/settings.gradle.kts ]; then
            SETTINGS_OK=1
          fi
          APP_DIR_OK=0
          if [ -d project/app ]; then
            APP_DIR_OK=1
          fi

          if [ "$MANIFEST_OK" -eq 0 ] || [ "$BUILD_GRADLE_OK" -eq 0 ] || [ "$SETTINGS_OK" -eq 0 ] || [ "$APP_DIR_OK" -eq 0 ]; then
            {
              echo "### ❌ Projet incomplet après extraction"
              echo ""
              echo "| Élément requis | Présent |"
              echo "|---|---|"
              echo "| AndroidManifest.xml | $([ $MANIFEST_OK -eq 1 ] && echo ✅ || echo ❌) |"
              echo "| build.gradle(.kts) | $([ $BUILD_GRADLE_OK -eq 1 ] && echo ✅ || echo ❌) |"
              echo "| settings.gradle(.kts) | $([ $SETTINGS_OK -eq 1 ] && echo ✅ || echo ❌) |"
              echo "| dossier app/ | $([ $APP_DIR_OK -eq 1 ] && echo ✅ || echo ❌) |"
            } >> "$GITHUB_STEP_SUMMARY"
            echo "::error::Projet Android incomplet, voir le résumé du job pour le détail."
            exit 1
          fi

          chmod +x project/gradlew 2>/dev/null || true

      - name: Analyse
        id: analyze
        run: |
          CONFIG=builds/build-config.json
          if [ ! -f "$CONFIG" ]; then
            echo '{}' > "$CONFIG"
          fi
          echo "build_type=$(jq -r '.buildType // "debug"' $CONFIG)" >> "$GITHUB_OUTPUT"
          echo "apk_name=$(jq -r '.apkName // "app"' $CONFIG)" >> "$GITHUB_OUTPUT"
          echo "version_name=$(jq -r '.versionName // empty' $CONFIG)" >> "$GITHUB_OUTPUT"
          echo "version_code=$(jq -r '.versionCode // empty' $CONFIG)" >> "$GITHUB_OUTPUT"
          cat "$CONFIG"

      - name: Installation Gradle
        uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: "17"
          cache: gradle

      - name: Installation Gradle
        working-directory: project
        run: |
          set -e
          NEED_WRAPPER=0
          if [ ! -f gradlew ] || [ ! -s gradle/wrapper/gradle-wrapper.jar ]; then
            NEED_WRAPPER=1
          fi
          if [ "$NEED_WRAPPER" -eq 1 ]; then
            echo "Wrapper Gradle manquant ou invalide : régénération automatique avec Gradle 8.7."
            curl -sL https://services.gradle.org/distributions/gradle-8.7-bin.zip -o /tmp/gradle.zip
            unzip -q /tmp/gradle.zip -d /tmp
            /tmp/gradle-8.7/bin/gradle wrapper --gradle-version 8.7 --distribution-type bin
          fi
          chmod +x gradlew

      - name: Compilation
        working-directory: project
        run: |
          set -o pipefail
          TASK="assembleDebug"
          EXTRA_ARGS=""
          if [ "${{ steps.analyze.outputs.build_type }}" = "release" ]; then
            TASK="assembleRelease"
          fi
          if [ -n "${{ steps.analyze.outputs.version_name }}" ]; then
            EXTRA_ARGS="$EXTRA_ARGS -PversionNameOverride=${{ steps.analyze.outputs.version_name }}"
          fi
          if [ -n "${{ steps.analyze.outputs.version_code }}" ]; then
            EXTRA_ARGS="$EXTRA_ARGS -PversionCodeOverride=${{ steps.analyze.outputs.version_code }}"
          fi

          set +e
          ./gradlew $TASK $EXTRA_ARGS --no-daemon --stacktrace 2>&1 | tee ../gradle-output.log
          exit_code=${PIPESTATUS[0]}
          set -e

          if [ $exit_code -ne 0 ]; then
            {
              echo "### ❌ Échec de la compilation Gradle (code $exit_code)"
              echo ""
              echo "#### Erreurs détectées"
              echo '```'
              grep -E '^e: |: error:|FAILURE:|What went wrong|Execution failed for' ../gradle-output.log | head -n 60
              echo '```'
              echo ""
              echo "#### Fin du log (contexte)"
              echo '```'
              tail -n 40 ../gradle-output.log
              echo '```'
            } >> "$GITHUB_STEP_SUMMARY"
            echo "::error::La compilation Gradle a échoué. Voir le résumé du job pour les erreurs précises."
            exit $exit_code
          fi

      # Étape optionnelle : ne s'exécute que si les secrets de signature
      # sont configurés dans le dépôt (Settings > Secrets and variables > Actions).
      - name: Signature
        if: steps.analyze.outputs.build_type == 'release'
        working-directory: project
        env:
          KEYSTORE_BASE64: ${{ secrets.KEYSTORE_BASE64 }}
          KEYSTORE_PASSWORD: ${{ secrets.KEYSTORE_PASSWORD }}
          KEY_ALIAS: ${{ secrets.KEY_ALIAS }}
          KEY_PASSWORD: ${{ secrets.KEY_PASSWORD }}
        run: |
          if [ -z "$KEYSTORE_BASE64" ]; then
            echo "Aucun secret de signature configuré, l'APK release reste non signée."
            exit 0
          fi
          echo "$KEYSTORE_BASE64" | base64 -d > release.keystore
          APK=$(find app/build/outputs/apk -name "*.apk" | head -n1)
          $ANDROID_HOME/build-tools/*/apksigner sign \
            --ks release.keystore \
            --ks-pass pass:"$KEYSTORE_PASSWORD" \
            --ks-key-alias "$KEY_ALIAS" \
            --key-pass pass:"$KEY_PASSWORD" \
            "$APK"

      - name: Création APK
        id: apk
        working-directory: project
        run: |
          APK_PATH=$(find app/build/outputs/apk -name "*.apk" | head -n1)
          if [ -z "$APK_PATH" ]; then
            echo "::error::Compilation réussie mais aucune APK trouvée dans app/build/outputs/apk."
            exit 1
          fi
          NAME="${{ steps.analyze.outputs.apk_name }}.apk"
          cp "$APK_PATH" "$NAME"
          echo "apk_path=project/$NAME" >> "$GITHUB_OUTPUT"

      - name: Publication
        uses: actions/upload-artifact@v4
        with:
          name: ${{ steps.analyze.outputs.apk_name }}
          path: ${{ steps.apk.outputs.apk_path }}
          retention-days: 30
